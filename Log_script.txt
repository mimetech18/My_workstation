Core and Distribution Router (Single Router)

/system logging action add name=logserver remote=103.172.29.74  remote-log-format=syslog syslog-facility=local5 target=remote
/system logging set 0 topics=info,!firewall
/system logging add action=logserver topics=firewall

/ip firewall mangle add action=log chain=prerouting connection-state=established protocol=tcp src-address=10.0.0.0/8 tcp-flags=fin

/ip service set api port=1122
